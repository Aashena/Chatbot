conda activate chat_bot
conda install pip

  langchain \
  langchain-community \
  langchain-google-genai
  

pip install sentence-transformers #Only if you want to use a local embedding generation! it is not needed if you use cloud vector database
pip install fastapi
pip install uvicorn
pip install requests
pip install crawl4ai
playwright install

export TELEGRAM_BOT_TOKEN=$telegram bot token$ #This is for telegram notification
export TELEGRAM_CHAT_ID= $telegram chat id$ #This is for telegram notification

cd Documents/workplace/ChatBot
jupyter notebook

For testing the FastAPI:
uvicorn main:app --reload
To test the API:
http://localhost:8000/docs

To export the requirements.txt file for only the packages that we imported in our files:
pip install pipreqs && pipreqs .

Making the docker image after creating the Dockerfile:
docker build --platform linux/amd64 \ #to make it compatible with google cloud cpu architechture
  -t chatbot .

Running the docker image:
docker run -p 8080:8080 \
  --platform linux/amd64 \
  -e GOOGLE_API_KEY=your_key \
  -e UPSTASH_VECTOR_REST_URL=upstash_url \
  -e UPSTASH_VECTOR_REST_TOKEN=upstash_key \
  -e TELEGRAM_BOT_TOKEN=bot_token \
  -e TELEGRAM_CHAT_ID = chat_id \
  chatbot
