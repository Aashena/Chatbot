conda install pip

  langchain \

  langchain-google-genai \
  chromadb \

pip install sentence-transformers

cd Documents/workplace/ChatBot

For testing the FastAPI:
uvicorn main:app --reload
To test the API:
http://localhost:8000/docs

To export the requirements.txt file for only the packages that we imported in our files:
pip install pipreqs && pipreqs .

Making the docker image after creating the Dockerfile:
docker build --platform linux/amd64 \ #to make it compatible with google cloud cpu architechture
  -t chatbot .

Running the docker image:
docker run -p 8080:8080 \
  --platform linux/amd64 \
  -e GOOGLE_API_KEY=your_key \
  -e HF_HUB_OFFLINE=1 \
  -e TRANSFORMERS_OFFLINE=1 \
  chatbot
